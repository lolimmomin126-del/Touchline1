    local function RefreshZZZ()
        if getgenv().ZZZZHelperConnections.Parts then
            for _, helper in ipairs(getgenv().ZZZZHelperConnections.Parts) do
                if helper and helper.Parent then helper:Destroy() end
            end
        end
        if getgenv().ZZZZHelperConnections.Connections then
            for _, conn in ipairs(getgenv().ZZZZHelperConnections.Connections) do
                if conn and conn.Disconnect then conn:Disconnect() end
            end
        end

        getgenv().ZZZZHelperConnections.Parts = {}
        getgenv().ZZZZHelperConnections.Connections = {}

        if not Config.ZZZ then return end
        local footballs = workspace:FindFirstChild("Footballs")
        if not footballs then return end

        local function createHelperForBall(ball)
            if not ball:IsA("BasePart") then return end

            local helper = Instance.new("Part")
            helper.Name = "AirDribbleHelper"
            helper.Size = Vector3.new(Config.ZZZSize, 0.5, Config.ZZZSize)
            helper.Anchored = true
            helper.CanCollide = true
            helper.CanTouch = true
            helper.CanQuery = true
            helper.Transparency = Config.ZZZVis and 0.4 or 1
            helper.Material = Enum.Material.Neon
            helper.Color = Color3.fromRGB(139, 92, 246)
            helper.Parent = workspace

            table.insert(getgenv().ZZZZHelperConnections.Parts, helper)

            local conn = RunService.RenderStepped:Connect(function()
                if not helper or not helper.Parent or not ball or not ball.Parent then return end
                helper.Size = Vector3.new(Config.ZZZSize, 0.5, Config.ZZZSize)
                helper.Transparency = Config.ZZZVis and 0.4 or 1
                helper.CFrame = CFrame.new(ball.Position - Vector3.new(0, Config.ZZZOffset, 0))
            end)

            table.insert(getgenv().ZZZZHelperConnections.Connections, conn)
        end

        for _, child in ipairs(footballs:GetChildren()) do
            createHelperForBall(child)
        end

        table.insert(
            getgenv().ZZZZHelperConnections.Connections,
            footballs.ChildAdded:Connect(createHelperForBall)
        )
    end

        lbl.TextSize = 10
        lbl.TextXAlignment = Enum.TextXAlignment.Left
        lbl.Parent = f

        local val = Instance.new("TextLabel")
        val.Size = UDim2.new(0.35, 0, 0, 16)
        val.Position = UDim2.new(0.65, -10, 0, 3)
        val.BackgroundTransparency = 1
        val.Text = tostring(default) .. suffix
        val.TextColor3 = Color3.fromRGB(139, 92, 246)
        val.Font = Enum.Font.Code
        val.TextSize = 10
        val.TextXAlignment = Enum.TextXAlignment.Right
        val.Parent = f

        local bar = Instance.new("Frame")
        bar.Size = UDim2.new(1, -20, 0, 5)
        bar.Position = UDim2.new(0, 10, 0, 22)
        bar.BackgroundColor3 = Color3.fromRGB(22, 22, 28)
        bar.BorderSizePixel = 0
        bar.Parent = f

        local fill = Instance.new("Frame")
        fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
        fill.BackgroundColor3 = Color3.fromRGB(139, 92, 246)
        fill.BorderSizePixel = 0
        fill.Parent = bar

        local sliding = false
        local function update(input)
            local pct = math.clamp((input.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
            local v = math.floor(min + (max - min) * pct)
            fill.Size = UDim2.new(pct, 0, 1, 0)
            val.Text = tostring(v) .. suffix
            callback(v)
        end

        bar.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                sliding = true update(input)
            end
        end)
        UserInputService.InputChanged:Connect(function(input)
            if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then update(input) end
        end)
        UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then sliding = false end
        end)
    end

    local hue = 0
    local hbConn = RunService.Heartbeat:Connect(function()
        pcall(function()
            local char = LocalPlayer.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            local hum = char and char:FindFirstChildOfClass("Humanoid")

            hue = (hue + 0.005) % 1
            local rainbowColor = Color3.fromHSV(hue, 0.8, 1)

            if Config.ShowHUD then
                local fps = math.floor(1 / RunService.RenderStepped:Wait())
                local ping = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue())
                HUD.Text = "touchline.lua | " .. fps .. " fps | " .. ping .. "ms"
            end

            if hum then
                if Config.Speed then hum.WalkSpeed = Config.SpeedVal end
                if Config.Jump then hum.JumpPower = Config.JumpVal end
                if Config.AntiRagdoll and hum:GetState() == Enum.HumanoidStateType.Physics then hum:ChangeState(Enum.HumanoidStateType.GettingUp) end
                if Config.Bhop and hum.FloorMaterial ~= Enum.Material.Air and UserInputService:IsKeyDown(Enum.KeyCode.Space) then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
            end

            if Config.Airwalk and root then AirwalkPart.CFrame = CFrame.new(root.Position.X, root.Position.Y - 3.5, root.Position.Z) end
            if Config.Spinbot and root then root.CFrame = root.CFrame * CFrame.Angles(0, math.rad(30), 0) end

            if Config.Noclip and char then
                for _, p in ipairs(char:GetDescendants()) do if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false end end
            end

            if Config.Reach and root then
                local reachVec = Vector3.new(Config.ReachSize, Config.ReachSize, Config.ReachSize)
                local sphereColor = Config.RainbowReach and rainbowColor or Color3.fromRGB(139, 92, 246)

                for part in pairs(trackedBalls) do
                    if part.Parent then
                        part.Size = reachVec
                        part.Transparency = Config.ReachTrans
                        part.Material = Enum.Material.ForceField
                        part.Color = sphereColor
                        part.CanCollide = false

                        local dist = (part.Position - root.Position).Magnitude
                        if (Config.AutoTouch or Config.AutoShoot) and dist <= (Config.ReachSize + 2) then
                            firetouchinterest(root, part, 0)
                            firetouchinterest(root, part, 1)
                        end
                    end
                end

                local nearest, closeDist = nil, math.huge
                for part in pairs(trackedBalls) do
                    if part.Parent then
                        local d = (part.Position - root.Position).Magnitude
                        if d < closeDist then closeDist = d nearest = part end
                    end
                end

                if nearest and closeDist <= 100 then
                    reachSphere.CFrame = nearest.CFrame
                    reachSphere.Size = reachVec
                    reachSphere.Transparency = Config.ReachTrans
                    reachSphere.Color = sphereColor
                else
                    reachSphere.CFrame = CFrame.new(0, -1000, 0)
                end
            else
                for part in pairs(trackedBalls) do
                    if part.Parent and originalSizes[part] then
                        part.Size = originalSizes[part]
                        part.Transparency = 0
                        part.Material = Enum.Material.Plastic
                        part.CanCollide = true
                    end
                end
                reachSphere.CFrame = CFrame.new(0, -1000, 0)
            end

            for part in pairs(trackedBalls) do
                if part.Parent then
                    if Config.BypassBallHide then part.Transparency = 0 end
                    if Config.StickyDribble and root then part.CFrame = root.CFrame * CFrame.new(0, -1, -3)
                    elseif Config.Magnet and root then
                        if (part.Position - root.Position).Magnitude <= Config.MagnetDist then
                            part.CFrame = root.CFrame * CFrame.new(0, -1.2, -2)
                        end
                    end
                    if Config.AutoLookBall and root then
                        root.CFrame = CFrame.new(root.Position, Vector3.new(part.Position.X, root.Position.Y, part.Position.Z))
                    end
                end
            end
        end)
    end)

                    if helper and helper.Parent then helper:Destroy() end
                end
            end
            if getgenv().ZZZZHelperConnections.Connections then
                for _, conn in ipairs(getgenv().ZZZZHelperConnections.Connections) do
                    if conn and conn.Disconnect then conn:Disconnect() end
                end
            end
            getgenv().ZZZZHelperConnections = nil
        end

        ClearESP()

        Lighting.Brightness = OrigLighting.Bright
        Lighting.ClockTime = OrigLighting.Time
        Lighting.FogEnd = OrigLighting.Fog
        workspace.Gravity = OrigLighting.Grav

        for part in pairs(trackedBalls) do
            if part.Parent and originalSizes[part] then
                part.Size = originalSizes[part]
                part.Transparency = 0
                part.Material = Enum.Material.Plastic
                part.CanCollide = true
                part.Anchored = false
            end
        end

        if hum then hum.WalkSpeed = 16 hum.JumpPower = 50 hum.HipHeight = 0 Camera.CameraSubject = hum end

        if reachSphere then reachSphere:Destroy() end
        if AirwalkPart then AirwalkPart:Destroy() end

        Camera.FieldOfView = 70
        getgenv().UnloadTouchline = nil
    end
